```python
"""🎭 กติกาปาร์ตี้เทิร์นเบส — ตรรกะล้วน ไม่มี I/O ไม่มี DB ไม่รู้จักการวาดภาพ

แยกไว้ต่างหากเพื่อให้เทสง่ายและอ่านจบในไฟล์เดียว
ใครจะเปลี่ยนสมดุลเกมมาแก้ที่นี่ที่เดียว
"""

# ธาตุไหนข่มธาตุไหน — ยึดจากเกมหลัก (bot/game_modifiers.py)
STRONG = {
    "fire": "wood", "wood": "earth", "earth": "wind",
    "wind": "water", "water": "fire", "light": "dark", "dark": "light",
}
BONUS = 1.5      # ข่มได้ ดาเมจ x1.5
PENALTY = 0.7    # โดนข่ม ดาเมจ x0.7

# ชนิดการเล็ง -> เล็งฝั่งไหนได้บ้าง
_TARGET_SIDES = {
    "single": "enemy",   # ตัวเดียว ฝั่งตรงข้าม
    "all": "enemy",      # ทุกตัว ฝั่งตรงข้าม
    "ally": "same",      # ฝั่งเดียวกัน (ฮีล/ชุบ)
    "self": "same",      # ตัวเอง
}


def element_multiplier(attacker_el, target_el):
    """ตัวคูณดาเมจจากธาตุ — ไม่มีธาตุ/ธาตุแปลก = 1.0 (ไม่ได้ไม่เสีย)"""
    if not attacker_el or not target_el:
        return 1.0
    if STRONG.get(attacker_el) == target_el:
        return BONUS
    if STRONG.get(target_el) == attacker_el:
        return PENALTY
    return 1.0


def can_target(kind, actor_side, target_side):
    """สกิลชนิดนี้เล็งเป้าฝั่งนี้ได้ไหม

    actor_side/target_side = "party" หรือ "monster"
    ชนิดที่ไม่รู้จัก = เล็งไม่ได้ (กันคนยิง API ส่งค่ามั่ว)
    """
    want = _TARGET_SIDES.get(kind)
    if want is None:
        return False
    same = actor_side == target_side
    return same if want == "same" else not same
```

- [ ] **Step 6: รันยืนยันแดง แล้วเขียน `events.py`**

```python
"""🎭 เหตุการณ์ที่แกนส่งให้ renderer วาด

⚠️ นี่คือ "สัญญา" ระหว่างแกน Python กับ renderer ทุกตัว
   แก้รูปแบบตรงนี้ = renderer ทุกตัวพัง — แก้แล้วต้องไล่แก้ทุก renderer ด้วย

หลักการ: เซิร์ฟเวอร์บอกว่า "เกิดอะไรขึ้น" ไม่บอกว่า "วาดยังไง"
         renderer อ่านรายการนี้แล้วตีความเป็นภาพเอง
         → เปลี่ยนกราฟิก = เขียน renderer ใหม่ ไม่ต้องแตะแกนเลย
"""


def turn_start(who):
    return {"t": "turn_start", "who": who}


def dice(who, rolls):
    return {"t": "dice", "who": who, "rolls": list(rolls), "total": sum(rolls)}


def attack(frm, to, dmg, element, crit=False):
    return {"t": "attack", "from": frm, "to": to,
            "dmg": int(dmg), "element": element, "crit": bool(crit)}


def aoe(frm, targets, dmgs, element):
    """โจมตีหลายเป้าพร้อมกัน — จำนวนเป้ากับดาเมจต้องเท่ากัน"""
    targets, dmgs = list(targets), [int(d) for d in dmgs]
    if len(targets) != len(dmgs):
        raise ValueError(f"เป้า {len(targets)} ตัว แต่ดาเมจ {len(dmgs)} ค่า — ไม่ตรงกัน")
    return {"t": "aoe", "from": frm, "targets": targets,
            "dmgs": dmgs, "element": element}


def heal(frm, to, amount):
    return {"t": "heal", "from": frm, "to": to, "amount": int(amount)}


def die(who):
    return {"t": "die", "who": who}


def phase(who, phase_name):
    """บอสเปลี่ยนเฟส — renderer เอาไปเล่นอนิเมชันพิเศษ"""
    return {"t": "phase", "who": who, "phase": phase_name}
```

```python
"""🎭 ห้องปาร์ตี้ — นั่งบนล็อบบี้กลาง (bot/lobby.py)

⚠️ lobby.MAX_SEATS = 6 (ค่าของสนามรบ) แต่เจ้าของเกมเคาะว่าปาร์ตี้ 5 คน
   ชั้นนี้จึงต้องคุม 5 เอง ห้ามไปแก้ค่าในล็อบบี้กลาง (สนามรบใช้อยู่)
"""
from bot import lobby

GAME = "party"
MAX_PARTY = 5          # เจ้าของเกมเคาะ 09-03
MIN_TO_START = 2


def _room(uid):
    code = lobby.room_of(GAME, uid)
    return lobby.load(code) if code else None


def open_party(uid, name):
    """เปิดปาร์ตี้ใหม่ — คนเปิดเป็นหัวห้อง"""
    st = lobby.create(GAME, uid, {"uid": str(uid), "name": name},
                      {"phase": "lobby"})
    return True, st


def join_party(uid, name, code):
    """เข้าปาร์ตี้ด้วยรหัส — คืน (ok, ห้อง|ข้อความผิดพลาด)"""
    st = lobby.load(str(code or "").strip().upper())
    if st and st.get("game") == GAME and len(st["seats"]) >= MAX_PARTY:
        if not any(s["uid"] == str(uid) for s in st["seats"]):
            return False, {"error": f"🚪 ปาร์ตี้เต็มแล้ว (สูงสุด {MAX_PARTY} คน)"}
    return lobby.join(GAME, uid, code, {"uid": str(uid), "name": name})


def leave_party(uid):
    return lobby.leave(GAME, uid)


def begin(uid):
    """หัวห้องกดเริ่ม — ต้องมีอย่างน้อย 2 คน"""
    st = _room(uid)
    if not st:
        return False, {"error": "ยังไม่ได้อยู่ปาร์ตี้ไหน"}
    if st["host"] != str(uid):
        return False, {"error": "เฉพาะหัวห้องเท่านั้นที่เริ่มได้"}
    if len(st["seats"]) < MIN_TO_START:
        return False, {"error": f"ต้องมีอย่างน้อย {MIN_TO_START} คนถึงจะเริ่มได้"}
    st["started"] = True
    st["extra"]["phase"] = "battle"
    return True, lobby.save(st["room"], st)


def view(uid):
    """🔒 ประตูเดียวก่อนส่งขึ้นเบราว์เซอร์"""
    out = lobby.view(GAME, uid)
    st = _room(uid)
    if st:
        out["max"] = MAX_PARTY          # ทับค่า 6 ของล็อบบี้กลาง
        out["phase"] = st["extra"].get("phase", "lobby")
    return out
```
